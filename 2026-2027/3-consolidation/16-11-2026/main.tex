\section{Les dictionnaires}

\begin{frame}[fragile,t]{Regrouper les informations d'un joueur}
On veut modéliser un jeu à plusieurs joueurs, où les joueurs ont :

\begin{onlyenv}<1-2>
\begin{itemize}
  \item un pseudo
  \item une couleur
  \item un score
\end{itemize}
\end{onlyenv}
\begin{onlyenv}<2->
\vspace{-0.5em}
On peut bien sûr utiliser des listes :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\footnotesize]{python}
pseudos = ["aze89", "yoplait", "azareth"]
couleurs = ["red", "yellow", "green"]
scores = [0, 0, 0]
\end{minted}
\end{mdframed}
\end{onlyenv}
\begin{onlyenv}<3>
\vspace{-0.5em}
Mais :
\begin{itemize}
  \item il faut se trimballer trois listes
  \item et ne pas se planter quand on ajoute un joueur
  \item ou une caractéristique
\end{itemize}
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Groupons par joueur}
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\footnotesize]{python}
joueur1 = ["qsdf89", "red", 25]
joueur2 = ["yoplait", "yellow", 20]
joueur3 = ["azareth", "green", 28]
joueurs = [joueur1, joueur2, joueur3]
\end{minted}
\end{mdframed}

\begin{onlyenv}<2->
\vspace{-0.5em}
Mais il faut se rappeler que :
\begin{itemize}
  \item 0 correspond au pseudo
  \item 1 correspond à la couleur
  \item 2 correspond au score
\end{itemize}
\end{onlyenv}
\begin{onlyenv}<3>
On aimerait donc donner un nom à chaque indice !
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Les dictionnaires}
C'est précisément ce que font les dictionnaires !
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\scriptsize]{python}
joueur1 = {'pseudo': 'qsdf89',
           'couleur': 'red',
           'score': 25}
joueur2 = {'pseudo': 'yoplait',
           'couleur': 'yellow',
           'score': 20}
joueur3 = {'pseudo': 'azareth',
           'couleur': 'green',
           'score': 28}
joueurs = [joueur1, joueur2, joueur3]
\end{minted}
\end{mdframed}

\begin{onlyenv}<2>
Les dictionnaires permettent ainsi de rassembler plusieurs informations concernant une personne.
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Dictionnaire : définition}
\textbf{Dictionnaire} : objet associant une liste de \alert{clés} à des \alert{valeurs}.
\begin{itemize}
  \item Entrées/clés (\textit{keys}) : d'un type de base \textit{immutable}
  \item Valeurs associées (\textit{values}) : de n'importe quel type
  \item Les paires (clé, valeur) sont appelées \textit{objets} (\textit{items})
\end{itemize}

\begin{onlyenv}<2>
Un objet de type \mintinline{python}{dict} est :
\begin{itemize}
  \item une \textbf{collection}
  \item \textbf{mutable} (= modifiable)
  \item \textbf{hétérogène} (peut contenir des types différents)
  \item \textbf{itérable} (utilisable dans un \mintinline{python}{for})
\end{itemize}
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Créer, accéder, modifier}
\begin{block}{Créer un dictionnaire}
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\footnotesize]{python}
# Pour créer un dictionnaire vide
dictionnaire = dict()  # Ou bien
dictionnaire = {}
# Pour l'initialiser avec des valeurs :
# voir précédemment
\end{minted}
\end{mdframed}
\end{block}

\begin{onlyenv}<2->
\begin{block}{Accéder à une valeur via sa clé}
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\footnotesize]{python}
dictionnaire[cle]
\end{minted}
\end{mdframed}
\end{block}
\end{onlyenv}
\begin{onlyenv}<3>
\begin{block}{Modifier une valeur}
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\footnotesize]{python}
dictionnaire[cle] = valeur
\end{minted}
\end{mdframed}
\end{block}
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Exemple}
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
joueur1 = {'pseudo': 'qsdf89',
           'couleur': 'red',
           'score': 25}
print(joueur1['couleur'])  # red
joueur1['score'] += 3
print(joueur1['score'])  # 28
\end{minted}
\end{mdframed}

\begin{onlyenv}<2>
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
joueur1['ville'] = 'Champs'
print(joueur1)
# {'pseudo': 'qsdf89', 'couleur': 'red',
#  'score': 28, 'ville': 'Champs'}
\end{minted}
\end{mdframed}
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Les clés (\textit{keys})}
On peut accéder à l'ensemble des clés du dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
dictionnaire.keys()
\end{minted}
\end{mdframed}

\begin{onlyenv}<2->
On peut donc tester si une clé est dans le dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
if cle in dictionnaire.keys():
\end{minted}
\end{mdframed}
\end{onlyenv}
\begin{onlyenv}<3->
C'est la même chose d'écrire (et on préfère) :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
if cle in dictionnaire:
\end{minted}
\end{mdframed}
\end{onlyenv}
\begin{onlyenv}<4>
On peut aussi \alert{itérer} sur le dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
for cle in dictionnaire:
# C'est pareil que (mais on préfère la version courte) :
for cle in dictionnaire.keys():
\end{minted}
\end{mdframed}
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Les valeurs (\textit{values})}
On peut accéder à l'ensemble des valeurs du dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
dictionnaire.values()
\end{minted}
\end{mdframed}

\begin{onlyenv}<2->
On peut donc tester si une valeur est dans le dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
if valeur in dictionnaire.values():
\end{minted}
\end{mdframed}
\end{onlyenv}
\begin{onlyenv}<3>
On peut aussi \alert{itérer} sur \textit{les valeurs} du dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
for valeur in dictionnaire.values():
\end{minted}
\end{mdframed}
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Les objets (\textit{items})}
On peut accéder à l'ensemble des objets du dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
dictionnaire.items()
\end{minted}
\end{mdframed}

\begin{onlyenv}<2->
On peut donc tester si un couple (clé, valeur) est dans le dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}

# C'est pareil que (et on préfère) :
if dictionnaire[cle] == valeur:
\end{minted}
\end{mdframed}
\end{onlyenv}
\begin{onlyenv}<3>
On peut aussi \alert{itérer} sur \textit{les objets} du dictionnaire :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
for (cle, valeur) in dictionnaire.items():
\end{minted}
\end{mdframed}
\end{onlyenv}
\end{frame}
