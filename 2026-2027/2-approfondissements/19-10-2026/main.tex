\section{Les fonctions sur les listes}

\begin{frame}[fragile]{Mutabilité, immutabilité}

  \begin{block}{Rappel}
    \vspace{0pt}
  Une fonction \alert{ne peut pas modifier} une variable de type \mintinline{python}{int} donnée en argument :
\vspace{-0.5em}
  \begin{mdframed}[roundcorner=5pt]
  \begin{minted}{python}
def essaie_de_modifier_pour_voir(x):
    x = 5

x = 12
essaie_de_modifier_pour_voir(x)
print(x)
>>> 12
\end{minted}
\end{mdframed}

\vspace{-0.5em}
C'est aussi vrai pour tous les types ``primitifs'' : \newline \mintinline{python}{int}, \mintinline{python}{float}, \mintinline{python}{str}, \mintinline{python}{bool}.

\vspace{-0.5em}
On dit que les types primitifs sont \alert{immutables}.
\vspace{1em}
  \end{block}

  \end{frame}

\begin{frame}[fragile]{Mutabilité, immutabilité}
  Une fonction \alert{peut modifier} la liste donnée en argument :
  \begin{mdframed}[roundcorner=5pt]
  \begin{minted}{python}
def essaie_de_modifier_pour_voir_liste(lst):
    lst[0] = 5

lst = [12]
essaie_de_modifier_pour_voir_liste(lst)
print(lst)
>>> [5]
\end{minted}
\end{mdframed}

On dit que les listes sont \alert{mutables}.
  \end{frame}

\section{Les fonctions classiques sur les listes}

\begin{frame}[fragile,t]{Réaffecter n'est pas modifier}
Une fonction ne peut pas faire pointer \mintinline{python}{lst} vers une autre liste :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
def essaie_de_modifier_pour_voir_liste_bis(lst):
    lst = [5]

lst = [12]
essaie_de_modifier_pour_voir_liste_bis(lst)
print(lst)
>>> [12]
\end{minted}
\end{mdframed}

\begin{onlyenv}<2>
\begin{block}{Dans la fonction}
  \vspace{-0.25em}
  \begin{itemize}
    \item \mintinline{python}{lst = [5]} : \alert{réaffecter} le nom local, sans effet dehors
    \item \mintinline{python}{lst[0] = 5} : \alert{modifier} la liste elle-même, visible dehors
  \end{itemize}
\end{block}
\end{onlyenv}
\end{frame}

\begin{frame}[fragile,t]{Les chaînes sont immutables}

\end{frame}

\begin{frame}[fragile,t]{Exercice : étendre une liste}
\begin{onlyenv}<1>
\begin{block}{Exercice}
Une fonction qui ajoute tous les éléments de la seconde liste \alert{à la fin} de la première.
\end{block}
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\footnotesize]{python}
def etend_liste(une_liste, autre_liste):
    ...
\end{minted}
\end{mdframed}
\begin{itemize}
  \item Pas de \mintinline{python}{return} : \mintinline{python}{une_liste} est modifiée \alert{sur place}
  \item On ne doit pas modifier \mintinline{python}{autre_liste}
\end{itemize}
\end{onlyenv}
\begin{onlyenv}<2>
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
def etend_liste(une_liste, autre_liste):
    for elem in autre_liste:
        une_liste.append(elem)

lst = [3, 'toto', 4.5]
lst2 = [False, None]
etend_liste(lst, lst2)
print(lst)
>>> [3, 'toto', 4.5, False, None]
print(lst2)
>>> [False, None]
\end{minted}
\end{mdframed}
\end{onlyenv}
\begin{onlyenv}<3>
Cette fonctionnalité existe déjà :
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\small]{python}
une_liste.extend(autre_liste)
\end{minted}
\end{mdframed}
\end{onlyenv}
\end{frame}

  \item \mintinline{python}{i} trop grand : on insère à la fin

\begin{onlyenv}<2>
\begin{mdframed}[roundcorner=5pt]
\begin{minted}[fontsize=\scriptsize]{python}
def ajouter(lst, i, x):
    i = max(0, min(i, len(lst)))   # indice correct
    lst.append(None)                # une case vide à la fin
    j = len(lst) - 1
    while j > i:                    # décalage vers la droite
        lst[j] = lst[j - 1]
        j -= 1

print(lst)

\begin{frame}[fragile,t]{Récapitulatif : méthodes qui modifient la liste}
\footnotesize
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}ll@{}}
\textbf{méthode} & \textbf{effet} \\ \hline
\texttt{lst.append(x)} & ajoute \texttt{x} à la fin \\
\texttt{lst.extend(lst2)} & ajoute les éléments de \texttt{lst2} à la fin \\
\texttt{lst.insert(i, x)} & ajoute \texttt{x} à l'indice \texttt{i} \\
\texttt{lst.remove(x)} & retire la première occurrence de \texttt{x}$^{(1)}$ \\
\texttt{lst.pop()} & retire et renvoie le dernier élément$^{(2)}$ \\
\texttt{lst.pop(i)} & retire et renvoie l'élément d'indice \texttt{i}$^{(3)}$ \\
\texttt{lst.clear()} & vide la liste \\
\texttt{lst.sort()} & trie la liste$^{(4)}$ \\
\texttt{lst.reverse()} & renverse la liste \\
\end{tabular}
\end{center}
\scriptsize $^{(1)}$ erreur si \texttt{x} n'est pas dans \texttt{lst} \quad $^{(2)}$ si \texttt{lst} est vide \quad $^{(3)}$ si \texttt{i} n'est pas un indice \quad $^{(4)}$ si éléments incomparables

\end{frame}

\begin{frame}[fragile,t]{Récapitulatif : ce qui ne modifie pas la liste}
\footnotesize
\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{@{}ll@{}}
\textbf{méthode / opérateur} & \textbf{effet} \\ \hline
\texttt{lst.index(x)} & indice de la première occurrence de \texttt{x}$^{(1)}$ \\
\texttt{lst.count(x)} & nombre d'occurrences de \texttt{x} \\
\texttt{lst.copy()} & copie \emph{superficielle} de \texttt{lst} \\
\texttt{len(lst)} & longueur de \texttt{lst} \\
\texttt{min(lst), max(lst)} & plus petit, plus grand élément$^{(2)}$ \\
\texttt{sum(lst)} & somme des éléments (nombres) \\
\texttt{sorted(lst)} & copie triée de \texttt{lst}$^{(3)}$ \\
\texttt{lst1 + lst2} & concaténation (nouvelle liste) \\
\texttt{lst * n} & répétition (nouvelle liste) \\
\texttt{x in lst} & \texttt{True} si \texttt{x} apparaît dans \texttt{lst} \\
\end{tabular}
\end{center}
\scriptsize $^{(1)}$ erreur si \texttt{x} n'est pas dans \texttt{lst} \quad $^{(2)}$ si \texttt{lst} est vide ou contient des éléments incomparables \quad $^{(3)}$ si éléments incomparables

\end{frame}
