% ------------------------------------------------------------------
% Macros pour dessiner une liste (rangée de 8 cases de 1,3 cm)
% ------------------------------------------------------------------
% la rangée : #1,#2 = coin haut gauche
\newcommand{\rangee}[2]{%
  \draw[very thick, rounded corners=3pt] (#1,#2) rectangle ({#1+10.4},{#2-0.9});
  \foreach \i in {1,...,7}{\draw[thick] ({#1+1.3*\i},#2) -- ({#1+1.3*\i},{#2-0.9});}%
}
% contenu d'une case : #1,#2 = coin haut gauche de la rangée, #3 = numéro de case, #4 = élément
\newcommand{\caseval}[4]{\node[font=\ttfamily\scriptsize, text=eltcol] at ({#1+1.3*(#3)+0.65},{#2-0.45}) {#4};}
% indice sous une case
\newcommand{\caseidx}[3]{\node[font=\Large\bfseries, text=idxcol] at ({#1+1.3*(#3)+0.65},{#2-1.35}) {#3};}
% étiquette encadrée en jaune
\newcommand{\boite}[3]{\node[anchor=west, draw=cmgold, rounded corners=5pt, very thick, inner sep=5pt, font=\large] at (#1,#2) {#3};}
% puce : tiret bleu
\newcommand{\tiret}{\textcolor{cmblue}{\rule[0.35ex]{0.8em}{0.4ex}}\hspace{0.5em}}
% bulle (algorithme classique) : #1 = couleur, #2,#3 = centre, #4 = largeur, #5 = texte
\newcommand{\bulle}[5]{%
  \node[fill=#1, rounded corners=14pt, minimum width=#4cm, minimum height=0.9cm, text width={#4cm-0.3cm}, text=black!45, font=\scriptsize\bfseries, align=center] at (#2,#3) {#5};%
}

% ------------------------------------------------------------------
% Macros pour la partie « A revoir » (slices, listes de listes, tuples)
% ------------------------------------------------------------------
% rangée de #3 cases de 1,3 cm : #1,#2 = coin haut gauche
\newcommand{\rangeen}[3]{%
  \draw[very thick, rounded corners=3pt] (#1,#2) rectangle ({#1+1.3*#3},{#2-0.9});
  \foreach \i in {1,...,#3}{\ifnum\i<#3 \draw[thick] ({#1+1.3*\i},#2) -- ({#1+1.3*\i},{#2-0.9});\fi}%
}
% cadre jaune autour des cases #3 à #4 (incluses) d'une rangée dont le coin haut gauche est (#1,#2)
\newcommand{\selection}[4]{%
  \draw[cmgold, very thick, rounded corners=4pt] ({#1+1.3*(#3)-0.07},{#2+0.07}) rectangle ({#1+1.3*(#4+1)+0.07},{#2-0.97});%
}
% borne d'une tranche : trait vers le bas + texte ; #1 = abscisse, #2 = texte
\newcommand{\borne}[2]{%
  \draw[black!50] (#1,3.35) -- (#1,2.55);
  \node[anchor=north, font=\itshape\footnotesize, text=black!65] at (#1,2.55) {#2};%
}
% plateau de morpion : 3 x 3 cases de 1,05 cm, coin haut gauche en (0.5,5.3)
\newcommand{\plateau}{%
  \draw[very thick, rounded corners=3pt] (0.5,2.15) rectangle (3.65,5.3);

  \draw[very thick] (0.5,4.25) -- (3.65,4.25);%
}
\newcommand{\marque}[3]{\node[font=\Huge\bfseries, text=eltcol] at ({0.5+1.05*(#2+0.5)},{5.3-1.05*(#1+0.5)}) {#3};}
\newcommand{\lignep}[1]{\draw[cmgold, very thick, rounded corners=4pt] (0.4,{5.3-1.05*#1+0.07}) rectangle (3.75,{5.3-1.05*(#1+1)-0.07});}
